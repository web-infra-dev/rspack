import { rspackRequire } from "./runtime.mjs";
// ./index.js
const index_load = () => new Promise((resolve, reject) => {
	Promise.resolve(/*! require.ensure */).then((() => {
		Promise.resolve(/*! require.ensure */).then((async () => {
			const a = await import("./shared.mjs").then(m => m.a_namespaceObject);
			const b = await import("./shared.mjs").then(m => m.b_namespaceObject);
			resolve([a, b]);
		}).bind(null, rspackRequire))['catch'](error => reject(error));
	}).bind(null, rspackRequire))['catch'](error => reject(error));
});

it("should return each module namespace for imports nested in require.ensure", async () => {
	const [a, b] = await index_load();
	expect({ ...a }).toEqual({ default: "a", name: "module-a", id: 1 });
	expect({ ...b }).toEqual({ default: "b", name: "module-b", id: 2 });
});

export { index_load as load };

// The module cache
var moduleCache = {};
// The require function
function rspackRequire(moduleId) {
// Check if module is in cache
var cachedModule = moduleCache[moduleId];
if (cachedModule !== undefined) {
return cachedModule.exports;
}
// Create a new module (and put it into the cache)
var module = (moduleCache[moduleId] = {
exports: {}
});
// Execute the module function
modules[moduleId](module, module.exports, rspackRequire);

// Return the exports of the module
return module.exports;
}
export { rspackRequire };

// rspack/runtime/define_property_getters
var definePropertyGetters = (exports, getters, values) => {
	var define = (defs, kind) => {
		for(var key in defs) {
			if(hasOwnProperty(defs, key) && !hasOwnProperty(exports, key)) {
				Object.defineProperty(exports, key, { enumerable: true, [kind]: defs[key] });
			}
		}
	};
	define(getters, "get");
	define(values, "value");
};;
export { definePropertyGetters };
// rspack/runtime/has_own_property
var hasOwnProperty = (obj, prop) => (Object.prototype.hasOwnProperty.call(obj, prop));
// rspack/runtime/make_namespace_object
// define __esModule on exports
var makeNamespaceObject = (exports) => {
	if(typeof Symbol !== 'undefined' && Symbol.toStringTag) {
		Object.defineProperty(exports, Symbol.toStringTag, { value: 'Module' });
	}
	Object.defineProperty(exports, '__esModule', { value: true });
};;
export { makeNamespaceObject };

import { rspackRequire, definePropertyGetters, makeNamespaceObject } from "./runtime.mjs";
// NAMESPACE OBJECT: ./a.js
var a_namespaceObject = {};
makeNamespaceObject(a_namespaceObject);
definePropertyGetters(a_namespaceObject, { 
  "default": () => (a),
  id: () => (id),
  name: () => (a_name) });


// NAMESPACE OBJECT: ./b.js
var b_namespaceObject = {};
makeNamespaceObject(b_namespaceObject);
definePropertyGetters(b_namespaceObject, { 
  "default": () => (b),
  id: () => (b_id),
  name: () => (b_name) });


// ./a.js
/* export default */ const a = ('a');
const a_name = 'module-a';
const id = 1;

// ./b.js
/* export default */ const b = ('b');
const b_name = 'module-b';
const b_id = 2;

export { a_namespaceObject, b_namespaceObject };
