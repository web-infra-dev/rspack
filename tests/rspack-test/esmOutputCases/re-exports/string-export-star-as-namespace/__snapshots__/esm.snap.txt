```mjs title=foo_js.mjs
// ./foo.js
const someImport = 42;
const named = "hello";



export { named, someImport as "some import" };

```

```mjs title=main.mjs
export * as "all the stuff" from "./foo_js.mjs";

// ./index.js



it("should support string export names on namespace re-exports", async () => {
  const mod = await import(/* webpackIgnore: true */ "./main.mjs");

  expect(mod["some export"]).toBe(42);
  expect(mod["all the stuff"]["some import"]).toBe(42);
  expect(mod["all the stuff"].named).toBe("hello");
});

export { "some import" as "some export" } from "./foo_js.mjs";

```
