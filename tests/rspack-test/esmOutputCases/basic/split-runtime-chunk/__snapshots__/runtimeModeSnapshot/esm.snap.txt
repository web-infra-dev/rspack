```mjs title=main.mjs
import { rspackRequire, publicPath, scriptNonce, setPublicPath, setScriptNonce } from "./runtime.mjs";
import "./other.mjs";

// ./runtime-writes.js
setPublicPath(('/assets/'))
scriptNonce ?? setScriptNonce('nonce')
scriptNonce || setScriptNonce('fallback')
scriptNonce && setScriptNonce('nonce-updated')
setPublicPath(publicPath + ('nested/'))

// ./index.js
const other = rspackRequire("./other.js");



it('should have correct value', () => {
  const { rspackPublicPath } = ({rspackPublicPath: publicPath})
  expect((0,other.U)()).toBe(42)
  expect(publicPath).toBe('/assets/nested/')
  expect(rspackPublicPath).toBe('/assets/nested/')
  expect(rspackRequire.custom).toBeUndefined()
  expect(scriptNonce).toBe('nonce-updated')
})

export {};
```

```mjs title=runtime.mjs

var modules = {};
// The module cache
var moduleCache = {};
// The require function
function rspackRequire(moduleId) {
// Check if module is in cache
var cachedModule = moduleCache[moduleId];
if (cachedModule !== undefined) {
return cachedModule.exports;
}
// Create a new module (and put it into the cache)
var module = (moduleCache[moduleId] = {
exports: {}
});
// Execute the module function
modules[moduleId](module, module.exports, rspackRequire);

// Return the exports of the module
return module.exports;
}
export { rspackRequire };
// expose the modules object (modules)
var moduleFactories = modules;
export { moduleFactories };

var publicPath, scriptNonce;
// rspack/runtime/esm_register_module
moduleFactories.add = function registerModules(modules) { Object.assign(moduleFactories, modules) }
;
// rspack/runtime/nonce
var scriptNonce = undefined;;
export { scriptNonce };
function setScriptNonce(value) { return scriptNonce = value; }
export { setScriptNonce };
// rspack/runtime/public_path
var publicPath = "";;
export { publicPath };
function setPublicPath(value) { return publicPath = value; }
export { setPublicPath };
```
