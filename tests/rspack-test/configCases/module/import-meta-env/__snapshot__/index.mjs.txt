import {fileURLToPath as __rspack_fileURLToPath} from "url";

it("should work import.meta.env with EnvironmentPlugin", () => {
	expect({"AAA":"aaa","WEBPACK_API_URL":"https://api.example.com","NODE_ENV":"production"}.AAA).toBe("aaa");
});

it("should work import.meta.env with DotenvPlugin", () => {
	expect({"AAA":"aaa","WEBPACK_API_URL":"https://api.example.com","NODE_ENV":"production"}.WEBPACK_API_URL).toBe("https://api.example.com");
});
